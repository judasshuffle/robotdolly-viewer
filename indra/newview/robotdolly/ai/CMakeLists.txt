cmake_minimum_required(VERSION 3.16)
project(robotdolly_ai_control_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/EHsc)
endif()
enable_testing()
add_library(ai_control viewport.cpp)
if(WIN32)
  target_compile_definitions(ai_control PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(ai_control PUBLIC user32 gdi32 dwmapi)
  add_executable(mock_renderer mock_renderer.cpp)
  target_compile_definitions(mock_renderer PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(mock_renderer PRIVATE user32)
endif()
add_executable(ai_control_test control_test.cpp)
target_link_libraries(ai_control_test PRIVATE ai_control)
if(WIN32)
  add_test(NAME lifecycle COMMAND ai_control_test $<TARGET_FILE:mock_renderer>)
else()
  add_test(NAME unsupported_platform COMMAND ai_control_test)
endif()
